import java.util.*;

public class Main {

    public static boolean isLapindrome(String s) {
        int n = s.length();
        int mid = n / 2;

        int[] left = new int[26];
        int[] right = new int[26];

        // Count first half
        for (int i = 0; i < mid; i++) {
            left[s.charAt(i) - 'a']++;
        }

        // Count second half
        int start = (n % 2 == 0) ? mid : mid + 1;

        for (int i = start; i < n; i++) {
            right[s.charAt(i) - 'a']++;
        }

        // Compare frequencies
        return Arrays.equals(left, right);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int T = sc.nextInt();

        while (T-- > 0) {
            String s = sc.next();

            if (isLapindrome(s)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }

        sc.close();
    }
}
