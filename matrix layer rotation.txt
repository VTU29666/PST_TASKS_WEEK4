import java.io.*;
import java.util.*;
import java.util.stream.*;
import static java.util.stream.Collectors.toList;

class Result {

    /*
     * Complete the 'matrixRotation' function below.
     *
     * The function accepts:
     *  1. 2D_INTEGER_ARRAY matrix
     *  2. INTEGER r
     */

    public static void matrixRotation(List<List<Integer>> matrix, int r) {

        int m = matrix.size();
        int n = matrix.get(0).size();

        // Number of layers
        int layers = Math.min(m, n) / 2;

        for (int layer = 0; layer < layers; layer++) {

            int top = layer;
            int bottom = m - 1 - layer;
            int left = layer;
            int right = n - 1 - layer;

            // Store the current layer
            List<Integer> layerElements = new ArrayList<>();

            // Top row: left -> right
            for (int j = left; j <= right; j++) {
                layerElements.add(matrix.get(top).get(j));
            }

            // Right column: top+1 -> bottom
            for (int i = top + 1; i <= bottom; i++) {
                layerElements.add(matrix.get(i).get(right));
            }

            // Bottom row: right-1 -> left
            for (int j = right - 1; j >= left; j--) {
                layerElements.add(matrix.get(bottom).get(j));
            }

            // Left column: bottom-1 -> top+1
            for (int i = bottom - 1; i > top; i--) {
                layerElements.add(matrix.get(i).get(left));
            }

            // Effective rotations
            int rotation = r % layerElements.size();

            /*
             * For anti-clockwise rotation,
             * start reading from 'rotation'.
             */
            int index = rotation;

            // Put values back into top row
            for (int j = left; j <= right; j++) {
                matrix.get(top).set(j, layerElements.get(index));
                index = (index + 1) % layerElements.size();
            }

            // Put values back into right column
            for (int i = top + 1; i <= bottom; i++) {
                matrix.get(i).set(right, layerElements.get(index));
                index = (index + 1) % layerElements.size();
            }

            // Put values back into bottom row
            for (int j = right - 1; j >= left; j--) {
                matrix.get(bottom).set(j, layerElements.get(index));
                index = (index + 1) % layerElements.size();
            }

            // Put values back into left column
            for (int i = bottom - 1; i > top; i--) {
                matrix.get(i).set(left, layerElements.get(index));
                index = (index + 1) % layerElements.size();
            }
        }

        // Print the rotated matrix
        for (List<Integer> row : matrix) {
            for (int j = 0; j < row.size(); j++) {
                System.out.print(row.get(j));

                if (j < row.size() - 1) {
                    System.out.print(" ");
                }
            }
            System.out.println();
        }
    }
}

public class Solution {

    public static void main(String[] args) throws IOException {

        BufferedReader bufferedReader =
                new BufferedReader(new InputStreamReader(System.in));

        String[] firstMultipleInput =
                bufferedReader.readLine()
                        .replaceAll("\\s+$", "")
                        .split(" ");

        int m = Integer.parseInt(firstMultipleInput[0]);
        int n = Integer.parseInt(firstMultipleInput[1]);
        int r = Integer.parseInt(firstMultipleInput[2]);

        List<List<Integer>> matrix = new ArrayList<>();

        IntStream.range(0, m).forEach(i -> {
            try {
                matrix.add(
                    Stream.of(
                        bufferedReader.readLine()
                            .replaceAll("\\s+$", "")
                            .split(" ")
                    )
                    .map(Integer::parseInt)
                    .collect(toList())
                );
            } catch (IOException ex) {
                throw new RuntimeException(ex);
            }
        });

        Result.matrixRotation(matrix, r);

        bufferedReader.close();
    }
}
